
we_frontier_war = {
    icon = county_conquest_cb group = conquest
    combine_into_one = no target_titles = all target_title_tier = county
    should_show_war_goal_subview = yes mutually_exclusive_titles = { always = yes }
    allowed_for_character = { is_landed = yes NOT = { government_has_flag = we_frontier_wilderness } }
    allowed_against_character = { NOT = { government_has_flag = we_frontier_wilderness } }
    valid_to_start = {
        scope:target = {
            has_variable = we_frontier_stage var:we_frontier_stage = 2
            we_frontier_access = { COLONIST = scope:attacker }
        }
    }
    should_invalidate = {
        NOT = { any_in_list = { list = target_titles var:we_frontier_stage = 2 holder = { OR = { this = scope:defender target_is_liege_or_above = scope:defender } } } }
    }
    cost = { prestige = 75 }
    on_declaration = { on_declared_war = yes }
    on_victory = {
        scope:attacker = { save_scope_as = we_frontier_conqueror }
        create_title_and_vassal_change = { type = conquest save_scope_as = we_frontier_change }
        every_in_list = {
            list = target_titles
            we_frontier_take_outpost = yes
        }
        resolve_title_and_vassal_change = scope:we_frontier_change
        scope:attacker = { add_prestige = 50 }
        add_truce_attacker_victory_effect = yes
    }
    on_white_peace = { scope:attacker = { add_prestige = -25 } add_truce_white_peace_effect = yes }
    on_defeat = { scope:attacker = { add_prestige = -75 } add_truce_attacker_defeat_effect = yes }
    on_invalidated = { }
    on_victory_desc = we_frontier_war_victory_desc
    on_white_peace_desc = we_frontier_war_white_peace_desc
    on_defeat_desc = we_frontier_war_defeat_desc
    on_invalidated_desc = we_frontier_war_invalidated_desc
    war_name = WE_FRONTIER_WAR_NAME
    cb_name = WE_FRONTIER_WAR_CB_NAME
    attacker_wargoal_percentage = 0.8
    max_defender_score_from_occupation = 150 max_attacker_score_from_occupation = 150
    interface_priority = 85 ai_score_mult = { value = 0.35 }
}
